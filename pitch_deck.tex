\begin{frame}{The Core Problem}
    \vspace{0.6cm}
    \begin{itemize}
        \setlength\itemsep{1.5em}
        \item \textbf{The Core Crisis:} Severe soil degradation threatens agriculture and food security nationwide across Ethiopia.
        \item \textbf{The Pain Point:} Traditional land assessment methods are painfully slow, manual, and entirely reactive meaning valuable topsoil is lost before damage is noticed.
    \end{itemize}
\end{frame}

% ─────────────────────────────────────────────────────────────────
% Slide 3: Our AI Solution
% ─────────────────────────────────────────────────────────────────
\begin{frame}{Our AI Solution}
    \vspace{0.6cm}
    \begin{itemize}
        \setlength\itemsep{1.5em}
        \item \textbf{Our AI Platform:} An advanced artificial intelligence system using graph-based deep learning and topological research.
        \item \textbf{What it Does:} Models nationwide topography as an interconnected web to predict soil erosion risks with high precision.
    \end{itemize}
\end{frame}

% ─────────────────────────────────────────────────────────────────
% Slide 4: Market Opportunity & Traction (Updated with Live Prototype)
% ─────────────────────────────────────────────────────────────────
\begin{frame}{Market Opportunity \& Traction}
    \vspace{0.4cm}
    \begin{itemize}
        \setlength\itemsep{1.1em}
        \item \textbf{Market Size:} Ministry of Agriculture, environmental protection authorities, and regional bureaus across Ethiopia.
        \item \textbf{Current Traction:} Developed and validated a robust Semi-Supervised Graph Convolutional Network (GCN) achieving \textbf{93.89\% accuracy} across 255,000+ coordinates in 5 Amhara Woredas.
        \item \textbf{Live Dashboard Prototype:} Deployed interactive 2D/3D platform running client-side GCN inference engine in real time.
    \end{itemize}
